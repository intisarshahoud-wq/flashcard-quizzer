14def normalize_answer(text: str) -> str: 

15 """Return a comparison-ready form of ``text``. 

16 

17 Normalization makes answer checking forgiving of the differences a human 

18 typist should not be punished for: surrounding whitespace, runs of internal 

19 whitespace, letter case, and Unicode forms that look identical but encode 

20 differently (for example a pre-composed ``e`` with an accent versus the 

21 same glyph built from a base letter plus a combining mark). 

22 

23 Args: 

24 text: Raw text typed by the user, or the ``back`` of a card. 

25 

26 Returns: 

27 The normalized string used for equality comparison. 

28 

29 Examples: 

30 >>> normalize_answer(" Secure SHELL ") 

31 'secure shell' 

32 """ 

33 collapsed = " ".join(text.split()) 

34 return unicodedata.normalize("NFKC", collapsed).casefold()

37@dataclass(frozen=True, slots=True) 

38class Flashcard: 

39 """A single question/answer pair. 

40 

41 Attributes: 

42 front: The prompt shown to the user. 

43 back: The expected answer. 

44 """ 

45 

46 front: str 

47 back: str 

48 

49 @property 

50 def key(self) -> str: 

51 """Return a stable identifier for this card. 

52 

53 The normalized ``front`` is used rather than the list index so that 

54 progress recorded in one session still matches the card after the deck 

55 has been reordered or extended. 

56 """ 

57 return normalize_answer(self.front) 

58 

59 def matches(self, answer: str) -> bool: 

60 """Return ``True`` if ``answer`` is an acceptable response. 

61 

62 Comparison is case-insensitive and whitespace-insensitive. A card whose 

63 ``back`` contains ``|`` accepts any of the pipe-separated alternatives, 

64 which lets a deck author write ``"def|def keyword"`` without needing a 

65 richer file format. 

66 

67 Args: 

68 answer: The text the user typed. 

69 

70 Returns: 

71 ``True`` when the answer matches the card, ``False`` otherwise. 

72 """ 

73 given = normalize_answer(answer) 

74 if not given: 

75 return False 

76 accepted = (normalize_answer(part) for part in self.back.split("|")) 

77 return any(given == option for option in accepted if option)

97@dataclass 

98class SessionStats: 

99 """Accumulates results and derives the end-of-session summary. 

100 

101 The engine appends to this object; the reporter only reads from it. Keeping 

102 the arithmetic here means the percentages in the terminal summary and in an 

103 exported file can never disagree. 

104 """ 

105 

106 results: list[CardResult] = field(default_factory=list) 

107 

108 def record(self, result: CardResult) -> None: 

109 """Append ``result`` to the session history.""" 

110 self.results.append(result) 

111 

112 @property 

113 def total_questions(self) -> int: 

114 """Return how many questions were answered.""" 

115 return len(self.results) 

116 

117 @property 

118 def correct_count(self) -> int: 

119 """Return how many answers were correct.""" 

120 return sum(1 for result in self.results if result.correct) 

121 

122 @property 

123 def incorrect_count(self) -> int: 

124 """Return how many answers were incorrect.""" 

125 return self.total_questions - self.correct_count 

126 

127 @property 

128 def accuracy(self) -> float: 

129 """Return accuracy as a percentage in the range 0.0-100.0. 

130 

131 An empty session scores ``0.0`` rather than raising, because quitting 

132 before the first answer is a normal way to end a quiz. 

133 """ 

134 if not self.results: 

135 return 0.0 

136 return self.correct_count / self.total_questions * 100.0 

137 

138 @property 

139 def missed_cards(self) -> list[Flashcard]: 

140 """Return the distinct cards answered incorrectly, in first-miss order.""" 

141 seen: set[str] = set() 

142 missed: list[Flashcard] = [] 

143 for result in self.results: 

144 if result.correct or result.card.key in seen: 

145 continue 

146 seen.add(result.card.key) 

147 missed.append(result.card) 

148 return missed 

149 

150 @property 

151 def total_seconds(self) -> float: 

152 """Return the total time spent answering.""" 

153 return sum(result.elapsed_seconds for result in self.results) 

154 

155 @property 

156 def average_seconds(self) -> float: 

157 """Return the mean seconds per question, or ``0.0`` for an empty session.""" 

158 if not self.results: 

159 return 0.0 

160 return self.total_seconds / self.total_questions 

161 

162 @property 

163 def fastest_result(self) -> CardResult | None: 

164 """Return the quickest correct answer, or ``None`` if there was none.""" 

165 correct = [result for result in self.results if result.correct] 

166 if not correct: 

167 return None 

168 return min(correct, key=lambda result: result.elapsed_seconds) 

169 

170 @property 

171 def longest_streak(self) -> int: 

172 """Return the length of the longest run of consecutive correct answers.""" 

173 best = 0 

174 current = 0 

175 for result in self.results: 

176 current = current + 1 if result.correct else 0 

177 best = max(best, current) 

178 return best
