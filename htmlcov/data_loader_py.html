1"""Loading and validation of flashcard decks from JSON files. 

2 

3Two on-disk shapes are supported, because both appear in the sample data: 

4 

5* **Array format** -- a bare list of card objects:: 

6 

7 [{"front": "API", "back": "Application Programming Interface"}] 

8 

9* **Object format** -- a wrapper with an optional deck name:: 

10 

11 {"name": "Python Basics", "cards": [{"front": "...", "back": "..."}]} 

12 

13Every failure raises :class:`FlashcardLoadError` with a message written for a 

14user rather than a developer, so the CLI can print it and exit without ever 

15showing a traceback. 

16"""

67def _lookup(raw: dict[str, Any], candidates: tuple[str, ...]) -> tuple[str, Any] | None: 

68 """Return the first ``(key, value)`` in ``raw`` matching ``candidates``. 

69 

70 The comparison ignores case and surrounding whitespace so that a deck 

71 written with ``"Front"`` or ``"BACK"`` still loads. 

72 

73 Args: 

74 raw: The card object straight from JSON. 

75 candidates: Accepted key spellings, lower-cased. 

76 

77 Returns: 

78 The matching key and its value, or ``None`` if no key matched. 

79 """ 

80 normalized = {str(key).strip().lower(): key for key in raw} 

81 for candidate in candidates: 

82 if candidate in normalized: 

83 original = normalized[candidate] 

84 return original, raw[original] 

85 return None

88def _require_text( 

89 raw: dict[str, Any], candidates: tuple[str, ...], index: int, side: str 

90) -> str: 

91 """Return the validated text for one side of a card. 

92 

93 Args: 

94 raw: The card object. 

95 candidates: Accepted key spellings for this side. 

96 index: The card's 0-based position, used in error messages. 

97 side: Either ``"front"`` or ``"back"``, used in error messages. 

98 

99 Returns: 

100 The stripped text value. 

101 

102 Raises: 

103 FlashcardLoadError: If the key is absent, is not a string, or is blank. 

104 """ 

105 found = _lookup(raw, candidates) 

106 if found is None: 

107 expected = " or ".join(f'"{name}"' for name in candidates) 

108 raise FlashcardLoadError( 

109 f"{_ordinal(index)} is missing its {side} side " 

110 f"(expected a {expected} field)." 

111 ) 

112 key, value = found 

113 if not isinstance(value, str): 

114 raise FlashcardLoadError( 

115 f'{_ordinal(index)} has a non-text {side} side: "{key}" is ' 

116 f"{type(value).__name__}, expected text." 

117 ) 

118 text = value.strip() 

119 if not text: 

120 raise FlashcardLoadError( 

121 f'{_ordinal(index)} has an empty {side} side ("{key}").' 

122 ) 

123 return text

126def _extract_card_list(payload: Any) -> list[Any]: 

127 """Return the raw card list from either supported top-level shape. 

128 

129 Args: 

130 payload: The decoded JSON value. 

131 

132 Returns: 

133 The list of raw card entries. 

134 

135 Raises: 

136 FlashcardLoadError: If the payload is neither a list nor an object with 

137 a ``cards`` list. 

138 """ 

139 if isinstance(payload, list): 

140 return payload 

141 if isinstance(payload, dict): 

142 found = _lookup(payload, ("cards",)) 

143 if found is None: 

144 raise FlashcardLoadError( 

145 'Deck object has no "cards" field. Use either a list of cards ' 

146 'or {"cards": [...]}.' 

147 ) 

148 _, value = found 

149 if not isinstance(value, list): 

150 raise FlashcardLoadError( 

151 f'"cards" must be a list of card objects, ' 

152 f"got {type(value).__name__}." 

153 ) 

154 return value 

155 raise FlashcardLoadError( 

156 f'Deck must be a list of cards or an object with a "cards" list, ' 

157 f"got {type(payload).__name__}." 

158 )

172def _build_card(raw: Any, index: int) -> Flashcard: 

173 """Validate one raw entry and return it as a :class:`Flashcard`. 

174 

175 Args: 

176 raw: One entry from the card list. 

177 index: Its 0-based position, used in error messages. 

178 

179 Returns: 

180 The validated card. 

181 

182 Raises: 

183 FlashcardLoadError: If the entry is not an object or either side fails 

184 validation. 

185 """ 

186 if not isinstance(raw, dict): 

187 raise FlashcardLoadError( 

188 f"{_ordinal(index)} is {type(raw).__name__}, expected an object " 

189 'like {"front": "...", "back": "..."}.' 

190 ) 

191 front = _require_text(raw, FRONT_KEYS, index, "front") 

192 back = _require_text(raw, BACK_KEYS, index, "back") 

193 return Flashcard(front=front, back=back)

196def _collect_warnings(cards: list[Flashcard]) -> list[str]: 

197 """Return non-fatal notes about ``cards``. 

198 

199 A duplicate question is legal JSON and the quiz still runs, so it is 

200 reported rather than raised. It is worth surfacing because the adaptive 

201 mode keys its history on the question text and would merge the two. 

202 """ 

203 warnings: list[str] = [] 

204 seen: dict[str, int] = {} 

205 for position, card in enumerate(cards, start=1): 

206 if card.key in seen: 

207 warnings.append( 

208 f'Duplicate question at card #{position}: "{card.front}" ' 

209 f"also appears at card #{seen[card.key]}." 

210 ) 

211 else: 

212 seen[card.key] = position 

213 return warnings

216def load_deck(path: str | Path) -> Deck: 

217 """Load, validate and return the deck stored at ``path``. 

218 

219 Args: 

220 path: Path to a JSON deck in either supported format. 

221 

222 Returns: 

223 The loaded :class:`Deck`. 

224 

225 Raises: 

226 FlashcardLoadError: If the file is missing, unreadable, not valid JSON, 

227 not a recognised deck shape, empty, or contains an invalid card. 

228 The message is safe to print directly to the user. 

229 """ 

230 try: 

231 payload = read_json(path) 

232 except FileHandlerError as exc: 

233 raise FlashcardLoadError(str(exc)) from exc 

234 

235 source = Path(path).expanduser().resolve() 

236 raw_cards = _extract_card_list(payload) 

237 if not raw_cards: 

238 raise FlashcardLoadError(f"{source.name} contains no cards.") 

239 

240 cards = [_build_card(raw, index) for index, raw in enumerate(raw_cards)] 

241 return Deck( 

242 name=_extract_deck_name(payload, source), 

243 cards=tuple(cards), 

244 source=source, 

245 warnings=tuple(_collect_warnings(cards)), 

246 )
